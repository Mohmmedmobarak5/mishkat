        // إنشاء قائمة الـ 100 دارس (عبد الله 1 إلى عبد الله 100)

        function create100Students() {

            let list = [];

            for (let i = 1; i <= 100; i++) {

                list.push({

                    id: "s" + i,

                    name: `عبد الله ${i}`,

                    weekCompleted: false,

                    completedAt: "— بانتظار الإتمام —",

                    topicsDone: 0,

                    wirdProgress: "0%"

                });

            }

            return list;

        }

 

        // استرجاع الطلاب من الذاكرة أو إنشاؤهم من جديد

        let students = JSON.parse(localStorage.getItem('mishkat_100_students'));

        if (!students || students.length !== 100 || !students[0].name.startsWith('عبد الله')) {

            students = create100Students();

            localStorage.setItem('mishkat_100_students', JSON.stringify(students));

        }

 

        // توليد خطة 100 أسبوع للكتب السبعة

        let curriculum = JSON.parse(localStorage.getItem('mishkat_curriculum'));

        if (!curriculum) {

            curriculum = [];

            for (let w = 1; w <= 100; w++) {

                let b1 = books[((w - 1) * 3) % books.length];

                let b2 = books[((w - 1) * 3 + 1) % books.length];

                let b3 = books[((w - 1) * 3 + 2) % books.length];

 

                curriculum.push({

                    weekNumber: w,

                    title: `الأسبوع ${w}`,

                    status: w === 1 ? 'APPROVED' : 'DRAFT',

                    topics: [

                        { order: 1, bookTitle: b1.title, driveUrl: b1.driveUrl, startPage: ((w - 1) * 5) + 1, endPage: ((w - 1) * 5) + 15, task: "استخراج القواعد الأساسية وأثرها السلوكي." },

                        { order: 2, bookTitle: b2.title, driveUrl: b2.driveUrl, startPage: ((w - 1) * 5) + 1, endPage: ((w - 1) * 5) + 15, task: "تطبيق المنزلة التزكوية ومحاسبة النفس." },

                        { order: 3, bookTitle: b3.title, driveUrl: b3.driveUrl, startPage: ((w - 1) * 4) + 1, endPage: ((w - 1) * 4) + 14, task: "صياغة التكليف الميداني للأسبوع." }

        function resetAllStudentsData() {

            if (confirm("هل تريد إعادة ضبط كشف الدارسين الـ 100 (عبد الله 1 إلى عبد الله 100) وتصفير إنجازاتهم؟")) {

        function checkUrlParams() {

            const params = new URLSearchParams(window.location.search);

            const userParam = params.get('user');

            const roleParam = params.get('role');

 

            if (userParam) {

                const uNum = parseInt(userParam);

                if (uNum >= 1 && uNum <= 100) {

                    currentStudentIndex = uNum - 1;

                    switchRole('student');

                }

            } else if (roleParam === 'admin') {

                switchRole('admin');

            }

        }

        // تفاعل الدارس

        function toggleStudentTopic(idx) {

            studentTopicState[idx] = !studentTopicState[idx];

            renderAll();

        }

 

        function studentFinalizeWeek() {

            if (studentTopicState.filter(Boolean).length < 3) {

                alert('يجب إتمام دراسة المحاور الثلاثة كاملة لتتمكن من اعتماد الأسبوع.');

                return;

            }

            const time = new Date().toLocaleTimeString('ar-SA');

            const curStudent = students[currentStudentIndex];

            alert(`✓ مبارك يا (${curStudent.name})! تم تسجيل إتمامك للأسبوع عند الساعة ${time}.\nتم قفل الأسبوع في سجلك وتحديث لوحة المدير.`);

           

            curStudent.weekCompleted = true;

            curStudent.completedAt = "اليوم - " + time;

            curStudent.topicsDone = 3;

            curStudent.wirdProgress = "100%";

            saveAll();

            renderAll();

        }

 

        // قرارات المدير

        function updateWeekStatus(weekNum, status) {

            curriculum.find(w => w.weekNumber === weekNum).status = status;

            saveAll();

            renderAll();

        }

 

        function toggleStudentLock(stuId) {

            const s = students.find(x => x.id === stuId);

            s.weekCompleted = !s.weekCompleted;

            s.completedAt = s.weekCompleted ? "2026/09/24 - 09:30 ص" : "— تم فك القفل من المدير —";

            saveAll();

            renderAll();

        }

 

        function editWeekPrompt(weekNum) {

            const w = curriculum.find(x => x.weekNumber === weekNum);

            const val = prompt(`تعديل بداية صفحات المحور الأول للأسبوع ${weekNum}:`, w.topics[0].startPage);

            if (val !== null) {

                w.topics[0].startPage = parseInt(val) || w.topics[0].startPage;

                w.topics[0].endPage = w.topics[0].startPage + 14;

                saveAll();

                renderAll();

            }

        }

 

        function switchRole(role) {

            document.getElementById('btn-role-admin').classList.toggle('active', role === 'admin');

            document.getElementById('btn-role-student').classList.toggle('active', role === 'student');

 

            if (role === 'student') {

                showTab('student-view', null);

                document.getElementById('view-title').innerText = "بوابة الدارس | خطة الأسبوع الحالي";

            } else {

                showTab('dashboard', document.querySelector('.sidebar-menu a'));

                document.getElementById('view-title').innerText = "لوحة الرقابة والمؤشرات الإدارية";

            }

        }

 

        function showTab(tabId, el) {

            document.querySelectorAll('.tab-view').forEach(t => t.classList.remove('active'));

            document.querySelectorAll('.sidebar-menu a').forEach(a => a.classList.remove('active'));

            document.getElementById('tab-' + tabId).classList.add('active');

            if (el) el.classList.add('active');

        }
